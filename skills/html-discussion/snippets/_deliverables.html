defp dispatch(%{"method" => "tools/call",
                "id" => id,
                "params" => %{"name" => name_str,
                              "arguments" => args}}, session) do
  tool_atom = String.to_existing_atom(name_str)
  call_id = session.call_id
  atomized_args = atomize_args(args)
  case MCPRuntime.dispatch(tool_atom, call_id, atomized_args) do
    {:ok, state}      -> json_rpc_success(id, %{state_version: state.version})
    {:error, reason}  -> json_rpc_tool_error(id, reason)
    {:error, _, det}  -> json_rpc_tool_error(id, det)
  end
end
